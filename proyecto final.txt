# ====== CLASES DEL DIAGRAMA UML ======
class Persona:
    def __init__(self, nombre, documento, telefono):
        self._nombre = nombre
        self.documento = documento
        self.telefono = telefono

    def get_nombre(self):
        return self._nombre

    def set_nombre(self, nombre):
        self._nombre = nombre


class Cliente(Persona):
    def __init__(self, nombre, documento, telefono, direccion):
        super().__init__(nombre, documento, telefono)
        self.direccion = direccion
        self.ventas = []


class Empleado(Persona):
    def __init__(self, nombre, documento, telefono, salario, cargo):
        super().__init__(nombre, documento, telefono)
        self._salario = salario
        self.cargo = cargo

    def get_salario(self):
        return self._salario

    def set_salario(self, salario):
        if salario >= 0:
            self._salario = salario

    def calcular_salario(self):
        return self._salario


class Carnicero(Empleado):
    def __init__(self, nombre, documento, telefono, salario, bono):
        super().__init__(nombre, documento, telefono, salario, "Carnicero")
        self.bono = bono

    def calcular_salario(self):
        return self.get_salario() + self.bono


class Cajero(Empleado):
    def __init__(self, nombre, documento, telefono, salario, caja):
        super().__init__(nombre, documento, telefono, salario, "Cajero")
        self.caja = caja


class CategoriaProducto:
    def __init__(self, codigo, nombre):
        self.codigo = codigo
        self.nombre = nombre
        self.productos = []


class ProductoCarnico:
    def __init__(self, codigo, nombre, precio, categoria):
        self.codigo = codigo
        self.nombre = nombre
        self._precio = precio
        self.categoria = categoria
        self.lotes = []
        categoria.productos.append(self)

    def get_precio(self):
        return self._precio

    def set_precio(self, precio):
        if precio > 0:
            self._precio = precio

    def stock(self):
        total = 0
        for lote in self.lotes:
            if lote.estado() == "Vigente":
                total += lote.kilos
        return round(total, 3)

    def descontar(self, kilos):
        if kilos <= 0 or kilos > self.stock():
            return False
        for lote in sorted(self.lotes, key=lambda l: l.vencimiento):
            if lote.estado() == "Vigente":
                retirar = min(kilos, lote.kilos)
                lote.kilos = round(lote.kilos - retirar, 3)
                kilos = round(kilos - retirar, 3)
                if kilos <= 0:
                    break
        return True


class LoteCarne:
    def __init__(self, codigo, producto, kilos, vencimiento):
        self.codigo = codigo
        self.producto = producto
        self.kilos = kilos
        self.vencimiento = vencimiento
        self.cuarto = None
        self.registros = []
        producto.lotes.append(self)

    def estado(self):
        if self.vencimiento < date.today():
            return "Vencido"
        return "Vigente"


class CuartoFrio:
    def __init__(self, nombre, capacidad, minimo, maximo):
        self.nombre = nombre
        self.capacidad = capacidad
        self.minimo = minimo
        self.maximo = maximo
        self.lotes = []

    def espacio(self):
        ocupacion = sum(lote.kilos for lote in self.lotes)
        return round(self.capacidad - ocupacion, 3)

    def almacenar(self, lote):
        if lote.kilos > self.espacio():
            return False
        self.lotes.append(lote)
        lote.cuarto = self
        return True


class RegControlFrio:
    def __init__(self, lote, temperatura):
        self.lote = lote
        self.temperatura = temperatura
        self.fecha = datetime.now()
        lote.registros.append(self)

    def generar_alerta(self):
        cuarto = self.lote.cuarto
        if cuarto.minimo <= self.temperatura <= cuarto.maximo:
            return "Dentro del rango configurado"
        return "ALERTA: fuera del rango configurado"


class Proveedor:
    def __init__(self, codigo, nombre, telefono):
        self.codigo = codigo
        self.nombre = nombre
        self.telefono = telefono
        self.ordenes = []


class DetalleOrdenCompra:
    def __init__(self, producto, kilos, costo):
        self.producto = producto
        self.kilos = kilos
        self.costo = costo

    def subtotal(self):
        return self.kilos * self.costo


class OrdenCompra:
    def __init__(self, codigo, proveedor):
        self.codigo = codigo
        self.proveedor = proveedor
        self.fecha = date.today()
        self.detalles = []
        proveedor.ordenes.append(self)

    def agregar(self, detalle):
        self.detalles.append(detalle)

    def total(self):
        return sum(d.subtotal() for d in self.detalles)


class BasculaDigital:
    def __init__(self):
        self.peso = 0

    def capturar_peso(self, kilos):
        self.peso = kilos  # Entrada manual: bascula simulada
        return self.peso


class DetalleVenta:
    def __init__(self, producto, kilos, precio):
        self.producto = producto
        self.kilos = kilos
        self.precio = precio

    def subtotal(self):
        return round(self.kilos * self.precio, 2)


class MetodoPago:
    def __init__(self, monto):
        self.monto = monto

    def procesar_pago(self):
        return False


class PagoEfectivo(MetodoPago):
    def __init__(self, monto, recibido):
        super().__init__(monto)
        self.recibido = recibido

    def procesar_pago(self):
        return self.recibido >= self.monto


class PagoDigital(MetodoPago):
    def __init__(self, monto, referencia):
        super().__init__(monto)
        self.referencia = referencia

    def procesar_pago(self):
        return self.referencia.strip() != ""  # Simulacion


class Factura:
    def __init__(self, numero, venta, pago):
        self.numero = numero
        self.venta = venta
        self.pago = pago

    def imprimir(self):
        print("\n========== FACTURA", self.numero, "==========")
        print("Fecha:", self.venta.fecha)
        print("Cliente:", self.venta.cliente.get_nombre())
        if self.venta.cajero is not None:
            print("Cajero:", self.venta.cajero.get_nombre())
        print("----------------------------------")
        for d in self.venta.detalles:
            print(f"{d.producto.nombre} - {d.kilos:g} kg x ${d.precio:,.0f}")
            print(f"   Subtotal: ${d.subtotal():,.0f}")
        print("----------------------------------")
        print(f"TOTAL: ${self.venta.total():,.0f}")
        if isinstance(self.pago, PagoEfectivo):
            print(f"Efectivo recibido: ${self.pago.recibido:,.0f}")
            print(f"Cambio: ${self.pago.recibido - self.pago.monto:,.0f}")
        else:
            print("Pago digital simulado. Referencia:", self.pago.referencia)
        print("==================================\n")


class Venta:
    def __init__(self, numero, cliente, cajero):
        self.numero = numero
        self.cliente = cliente
        self.cajero = cajero
        self.fecha = date.today()
        self.detalles = []
        self.factura = None
        self.pago = None

    def agregar(self, detalle):
        usados = detalle.kilos
        for d in self.detalles:
            if d.producto == detalle.producto:
                usados += d.kilos
        if usados > detalle.producto.stock():
            return False
        self.detalles.append(detalle)
        return True

    def total(self):
        return round(sum(d.subtotal() for d in self.detalles), 2)

    def confirmar(self, pago):
        if not self.detalles or not pago.procesar_pago():
            return False
        # Verificacion del inventario antes de realizar cambios
        for d in self.detalles:
            if sum(x.kilos for x in self.detalles if x.producto == d.producto) > d.producto.stock():
                return False
        for d in self.detalles:
            d.producto.descontar(d.kilos)
        self.pago = pago
        self.factura = Factura(self.numero, self, pago)
        self.cliente.ventas.append(self)
        return True

def cargar():
    
    
    anterior = os.path.join(os.getcwd(), "datos_carniceria.json")
    if not os.path.exists(ARCHIVO) and os.path.isfile(anterior):
        if os.path.abspath(anterior) != os.path.abspath(ARCHIVO):
            shutil.copy2(anterior, ARCHIVO)

    if not os.path.exists(ARCHIVO):
        catalogo_inicial()
        completar_datos_de_ejemplo()
        guardar()
        return

    with open(ARCHIVO, "r", encoding="utf-8") as archivo:
        datos = json.load(archivo)

    categorias_por_id = {}
    productos_por_id = {}
    lotes_por_id = {}
    clientes_por_doc = {}
    empleados_por_doc = {}
    proveedores_por_id = {}

    for x in datos["categorias"]:
        c = CategoriaProducto(x["id"], x["nombre"])
        categorias.append(c)
        categorias_por_id[c.codigo] = c

    for x in datos["productos"]:
        p = ProductoCarnico(x["id"], x["nombre"], x["precio"],
                           categorias_por_id[x["categoria"]])
        productos.append(p)
        productos_por_id[p.codigo] = p

    for x in datos["lotes"]:
        lote = LoteCarne(x["id"], productos_por_id[x["producto"]],
                         x["kilos"], date.fromisoformat(x["vencimiento"]))
        cuarto.almacenar(lote)
        lotes.append(lote)
        lotes_por_id[lote.codigo] = lote

    for x in datos["clientes"]:
        c = Cliente(x["nombre"], x["documento"], x["telefono"], x["direccion"])
        clientes.append(c)
        clientes_por_doc[c.documento] = c

    for x in datos["empleados"]:
        if x["cargo"] == "Carnicero":
            e = Carnicero(x["nombre"], x["documento"], x["telefono"],
                          x["salario"], x["bono"])
        else:
            e = Cajero(x["nombre"], x["documento"], x["telefono"],
                       x["salario"], x["caja"])
        empleados.append(e)
        empleados_por_doc[e.documento] = e

    for x in datos["proveedores"]:
        p = Proveedor(x["id"], x["nombre"], x["telefono"])
        proveedores.append(p)
        proveedores_por_id[p.codigo] = p

    for x in datos["compras"]:
        o = OrdenCompra(x["id"], proveedores_por_id[x["proveedor"]])
        o.fecha = date.fromisoformat(x["fecha"])
        for d in x["detalles"]:
            o.agregar(DetalleOrdenCompra(productos_por_id[d["producto"]],
                                        d["kilos"], d["costo"]))
        compras.append(o)

    for x in datos["ventas"]:
        cajero = empleados_por_doc.get(x["cajero"])
        v = Venta(x["numero"], clientes_por_doc[x["cliente"]], cajero)
        v.fecha = date.fromisoformat(x["fecha"])
        for d in x["detalles"]:
            v.detalles.append(DetalleVenta(productos_por_id[d["producto"]],
                                          d["kilos"], d["precio"]))
        if x["pago"]["tipo"] == "efectivo":
            v.pago = PagoEfectivo(v.total(), x["pago"]["recibido"])
        else:
            v.pago = PagoDigital(v.total(), x["pago"]["referencia"])
        v.factura = Factura(v.numero, v, v.pago)
        v.cliente.ventas.append(v)
        ventas.append(v)

    for x in datos["controles"]:
        r = RegControlFrio(lotes_por_id[x["lote"]], x["temperatura"])
        r.fecha = datetime.fromisoformat(x["fecha"])
        controles.append(r)

    # Si el usuario ya tenia productos guardados pero aun no habia creado
    # personas o proveedores, se agregan datos de prueba sin borrar nada.
    if completar_datos_de_ejemplo():
        guardar()

def texto_obligatorio(mensaje):
    while True:
        valor = input(mensaje).strip()
        if valor:
            return valor
        print("Este dato no puede quedar vacio.")


def numero(mensaje, minimo=0):
    while True:
        try:
            valor = float(input(mensaje).replace(",", "."))
            if valor >= minimo:
                return valor
        except ValueError:
            pass
        print("Ingrese un numero valido.")


def entero(mensaje, minimo=0):
    while True:
        try:
            valor = int(input(mensaje))
            if valor >= minimo:
                return valor
        except ValueError:
            pass
        print("Ingrese un entero valido.")


def fecha_vencimiento():
    while True:
        try:
            fecha = date.fromisoformat(input("Vencimiento (AAAA-MM-DD): "))
            if fecha >= date.today():
                return fecha
        except ValueError:
            pass
        print("Introduzca una fecha valida que no haya pasado.")


def elegir(lista, titulo):
    if not lista:
        print("No hay registros de", titulo)
        return None
    print("\n---", titulo.upper(), "---")
    for i, obj in enumerate(lista, 1):
        nombre = obj.get_nombre() if isinstance(obj, Persona) else obj.nombre
        print(i, "-", nombre)
    opcion = entero("Numero (0 cancelar): ")
    if opcion == 0:
        return None
    if 1 <= opcion <= len(lista):
        return lista[opcion - 1]
    print("Opcion no valida")
    return None


def buscar_producto():
    palabra = input("Buscar nombre o categoria (Enter = todos): ").strip().lower()
    encontrados = []
    for p in productos:
        if palabra in p.nombre.lower() or palabra in p.categoria.nombre.lower():
            print(f"{p.codigo:3} | {p.nombre:25} | ${p.get_precio():,.0f}/kg | {p.stock():g} kg")
            encontrados.append(p)
    if not encontrados:
        print("No hay coincidencias")
        return None
    codigo = entero("Codigo del producto (0 cancelar): ")
    for p in encontrados:
        if p.codigo == codigo:
            return p
    print("Codigo no encontrado")
    return None


def registrar_lote(producto, kilos):
    if kilos > cuarto.espacio():
        print("Espacio insuficiente en el cuarto frio")
        return False
    vencimiento = fecha_vencimiento()
    lote = LoteCarne(codigo_nuevo(lotes), producto, kilos, vencimiento)
    cuarto.almacenar(lote)
    lotes.append(lote)
    guardar()
    print("Lote", lote.codigo, "registrado")
    return True

    cajeros = [e for e in empleados if isinstance(e, Cajero)]
    cajero = elegir(cajeros, "cajeros") if cajeros else None
    venta = Venta(max([v.numero for v in ventas], default=0) + 1, cliente, cajero)
    while True:
        producto = buscar_producto()
        if producto is None:
            break
        kilos = bascula.capturar_peso(numero("Kilos vendidos: ", 0.001))
        detalle = DetalleVenta(producto, kilos, producto.get_precio())
        if venta.agregar(detalle):
            print(f"Agregado: {producto.nombre} - ${detalle.subtotal():,.0f}")
        else:
            print("Inventario insuficiente")
        if input("Agregar otro producto? (s/n): ").lower() != "s":
            break
    if not venta.detalles:
        print("Venta cancelada: no hay productos")
        return
    print(f"TOTAL: ${venta.total():,.0f}")
    print("1. Efectivo    2. Pago digital (simulado)")
    opcion = input("Forma de pago: ")
    if opcion == "1":
        pago = PagoEfectivo(venta.total(), numero("Efectivo recibido: "))
    elif opcion == "2":
        pago = PagoDigital(venta.total(), texto_obligatorio("Referencia: "))
    else:
        print("Forma de pago invalida")
        return
    if venta.confirmar(pago):
        ventas.append(venta)
        guardar()
        venta.factura.imprimir()
    else:
        print("No se pudo confirmar la venta (pago o stock insuficiente)")

                kilos = numero("Kilos para agregar: ", 0.001)

            nombre = texto_obligatorio("Nombre: ")
            documento = texto_obligatorio("Documento: ")
            if any(c.documento == documento for c in clientes):
                print("Ya existe un cliente con ese documento")

            telefono = texto_obligatorio("Telefono: ")

            clientes.append(Cliente(nombre, documento, telefono, direccion))

        elif op == "5":

            telefono = texto_obligatorio("Telefono: ")

            guardar()

            tipo = input("Tipo: ")
            if tipo not in ("1", "2"):
                print("Tipo no valido")
                continue
            nombre = texto_obligatorio("Nombre: ")
            documento = texto_obligatorio("Documento: ")
            if any(e.documento == documento for e in empleados):
                print("Ya existe un empleado con ese documento")
                continue
            telefono = texto_obligatorio("Telefono: ")
            salario = numero("Salario base: ")

                empleado = Cajero(nombre, documento, telefono, salario, entero("Numero de caja: ", 1))

            guardar()
            print("Empleado registrado")

        elif op == "7":
            proveedor = elegir(proveedores, "proveedores")
            if proveedor is None:
                continue
            producto = buscar_producto()
            if producto is None:
                continue
            kilos = numero("Kilos comprados: ", 0.001)
            costo = numero("Costo por kg: ", 0.01)
